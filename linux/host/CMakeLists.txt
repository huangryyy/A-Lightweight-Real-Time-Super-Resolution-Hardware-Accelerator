cmake_minimum_required(VERSION 3.16)
project(FSRCNN_X4_video LANGUAGES CXX)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
add_executable(FSRCNN_X4_video FSRCNN_X4_video.cpp)
target_compile_features(FSRCNN_X4_video PRIVATE cxx_std_17)
target_compile_options(FSRCNN_X4_video PRIVATE -O3 -Wall -Wextra -Wno-unused-function -Wno-missing-field-initializers)
target_include_directories(FSRCNN_X4_video PRIVATE ..)
target_link_libraries(FSRCNN_X4_video PRIVATE OpenMP::OpenMP_CXX Threads::Threads)
